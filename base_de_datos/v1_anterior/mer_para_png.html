  usuarios {
    int id_usuario PK
    varchar nombre
    varchar correo UK
    varchar telefono

    datetime fecha_registro
  }
  cotizaciones {
    int id_cotizacion PK
    int id_usuario FK
    int id_tipo_trabajo FK
    int id_tipo_cubierta FK
    int id_tipo_estructura FK
    int id_diseno FK
    int id_plastico FK
    decimal largo
    decimal ancho
    decimal altura

    enum origen
    decimal costo_materiales
    decimal costo_mano_obra
    decimal total
    datetime fecha
  }

    int id_material PK
